# 02. DDL — 테이블, 제약조건, 인덱스

## 1. CREATE TABLE

```sql
CREATE TABLE IF NOT EXISTS app.users (
    id          bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    email       text        NOT NULL UNIQUE,
    name        text        NOT NULL,
    age         int         CHECK (age >= 0),
    status      text        NOT NULL DEFAULT 'active',
    meta        jsonb       NOT NULL DEFAULT '{}'::jsonb,
    tags        text[]      NOT NULL DEFAULT '{}',
    created_at  timestamptz NOT NULL DEFAULT now(),
    updated_at  timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT users_status_chk CHECK (status IN ('active','inactive','deleted'))
);
```

### IDENTITY vs SERIAL
```sql
-- 권장: SQL 표준 IDENTITY (PG10+)
id bigint GENERATED ALWAYS AS IDENTITY      -- 수동 입력 불가 (OVERRIDING 필요)
id bigint GENERATED BY DEFAULT AS IDENTITY  -- 수동 입력 허용

-- 레거시: serial (내부적으로 sequence + default nextval)
id bigserial   -- 소유권/권한 관리가 번거로움

-- ALWAYS 컬럼에 강제 입력
INSERT INTO users (id, email) OVERRIDING SYSTEM VALUE VALUES (1, 'a@b.c');
```

### 생성 컬럼 (Generated Column, PG12+)
```sql
CREATE TABLE product (
    price     numeric NOT NULL,
    tax_rate  numeric NOT NULL DEFAULT 0.1,
    -- STORED만 지원 (VIRTUAL 미지원)
    total     numeric GENERATED ALWAYS AS (price * (1 + tax_rate)) STORED
);
```

### 테이블 복사
```sql
CREATE TABLE t2 (LIKE t1 INCLUDING ALL);        -- 구조만 (제약/인덱스/기본값 포함)
CREATE TABLE t2 (LIKE t1 INCLUDING DEFAULTS INCLUDING CONSTRAINTS);
CREATE TABLE t2 AS SELECT * FROM t1;            -- 구조+데이터 (제약 미포함)
CREATE TABLE t2 AS SELECT * FROM t1 WITH NO DATA;

-- 임시 테이블 (세션 종료 시 삭제)
CREATE TEMP TABLE tmp_x (id int);
CREATE TEMP TABLE tmp_x ON COMMIT DROP (id int);   -- 트랜잭션 종료 시 삭제
CREATE UNLOGGED TABLE fast_x (id int);             -- WAL 미기록, 크래시 시 소실
```

## 2. 제약조건 (Constraint)

```sql
-- 컬럼 레벨 / 테이블 레벨 둘 다 가능. 이름 붙이는 걸 권장 (에러 메시지 추적)
CREATE TABLE orders (
    id         bigint GENERATED ALWAYS AS IDENTITY,
    user_id    bigint NOT NULL,
    order_no   text   NOT NULL,
    amount     numeric(15,2) NOT NULL,
    period     daterange,

    CONSTRAINT orders_pk       PRIMARY KEY (id),
    CONSTRAINT orders_no_uk    UNIQUE (order_no),
    CONSTRAINT orders_amt_chk  CHECK (amount > 0),
    CONSTRAINT orders_user_fk  FOREIGN KEY (user_id) REFERENCES users(id)
                               ON DELETE CASCADE ON UPDATE RESTRICT,
    -- EXCLUDE: PostgreSQL 고유. 기간 겹침 방지 등
    CONSTRAINT orders_no_overlap EXCLUDE USING gist (user_id WITH =, period WITH &&)
);
```

### FK 동작 옵션
| 옵션 | 동작 |
|---|---|
| `NO ACTION` (기본) | 위반 시 에러 (제약 검사 지연 가능) |
| `RESTRICT` | 즉시 에러 (지연 불가) |
| `CASCADE` | 부모 삭제/변경 시 자식도 함께 |
| `SET NULL` | 자식 FK 컬럼을 NULL로 |
| `SET DEFAULT` | 자식 FK 컬럼을 DEFAULT로 |

```sql
-- 부분 SET NULL (PG15+)
FOREIGN KEY (a, b) REFERENCES p(a, b) ON DELETE SET NULL (b)
```

### 지연 제약 (Deferrable)
```sql
CONSTRAINT fk_x FOREIGN KEY (...) REFERENCES ... DEFERRABLE INITIALLY DEFERRED;

-- 트랜잭션 내에서 커밋 시점까지 검사 미룸 (순환 참조 삽입 시 유용)
BEGIN;
SET CONSTRAINTS ALL DEFERRED;
-- ... 작업 ...
COMMIT;  -- 여기서 검사
```

### NOT VALID — 무중단 제약 추가
```sql
-- 기존 데이터 검사 없이 즉시 추가 (테이블 풀스캔/락 회피)
ALTER TABLE t ADD CONSTRAINT c CHECK (x > 0) NOT VALID;
-- 나중에 별도로 검증 (ACCESS EXCLUSIVE 락 안 잡음)
ALTER TABLE t VALIDATE CONSTRAINT c;
```

## 3. ALTER TABLE

```sql
ALTER TABLE t ADD COLUMN c text;
ALTER TABLE t ADD COLUMN c text NOT NULL DEFAULT 'x';  -- PG11+ 즉시 완료 (재작성 없음)
ALTER TABLE t DROP COLUMN c;
ALTER TABLE t DROP COLUMN IF EXISTS c CASCADE;
ALTER TABLE t RENAME COLUMN a TO b;
ALTER TABLE t RENAME TO t2;

ALTER TABLE t ALTER COLUMN c TYPE bigint;                    -- 테이블 재작성
ALTER TABLE t ALTER COLUMN c TYPE int USING c::int;          -- 변환식 필요 시
ALTER TABLE t ALTER COLUMN c SET NOT NULL;
ALTER TABLE t ALTER COLUMN c DROP NOT NULL;
ALTER TABLE t ALTER COLUMN c SET DEFAULT now();
ALTER TABLE t ALTER COLUMN c DROP DEFAULT;

ALTER TABLE t ADD CONSTRAINT ... ;
ALTER TABLE t DROP CONSTRAINT c;
ALTER TABLE t SET SCHEMA app;
ALTER TABLE t OWNER TO app_user;
```

**운영 주의:** `ALTER TABLE`은 대부분 `ACCESS EXCLUSIVE` 락을 잡는다. 대용량 테이블에서는
`lock_timeout` 설정 후 실행할 것.
```sql
SET lock_timeout = '3s';
ALTER TABLE big_table ADD COLUMN c text;
```

## 4. 인덱스

```sql
CREATE INDEX idx_users_email ON users (email);
CREATE UNIQUE INDEX idx_users_email_uk ON users (email);

-- 무중단 생성 (쓰기 락 안 잡음. 트랜잭션 블록 내 불가, 느림)
CREATE INDEX CONCURRENTLY idx_x ON t (c);
DROP INDEX CONCURRENTLY idx_x;

-- 부분 인덱스 (PostgreSQL 고유, 매우 유용)
CREATE INDEX idx_active ON users (email) WHERE status = 'active';

-- 표현식 인덱스
CREATE INDEX idx_lower_email ON users (lower(email));
-- 쿼리도 동일 표현식이어야 사용됨
SELECT * FROM users WHERE lower(email) = 'a@b.c';

-- 복합 인덱스 (컬럼 순서가 중요: 좌측 접두사 규칙)
CREATE INDEX idx_a_b ON t (a, b);   -- (a), (a,b) 검색 가능. (b) 단독은 불가

-- 정렬 지정
CREATE INDEX idx_d ON t (created_at DESC NULLS LAST);

-- INCLUDE (커버링 인덱스, PG11+) — 인덱스 온리 스캔용
CREATE INDEX idx_c ON t (a) INCLUDE (b, c);

-- NULLS NOT DISTINCT (PG15+) — NULL을 중복으로 취급
CREATE UNIQUE INDEX ON t (a) NULLS NOT DISTINCT;
```

### 인덱스 타입
| 타입 | 용도 |
|---|---|
| `btree` (기본) | 등호/범위/정렬. 대부분의 경우 |
| `hash` | 등호 전용. PG10+ 부터 WAL 지원. 이점 미미 |
| `gin` | 배열, JSONB, 전문검색 (다중 값 포함) |
| `gist` | 기하, 범위 타입, 최근접 검색, EXCLUDE 제약 |
| `spgist` | 공간 분할 (텍스트 접두사, 비균등 데이터) |
| `brin` | 초대용량 + 물리 순서 상관관계 (시계열 로그). 크기 극소 |

```sql
CREATE INDEX idx_tags ON t USING gin (tags);
CREATE INDEX idx_meta ON t USING gin (meta jsonb_path_ops);
CREATE INDEX idx_ts   ON logs USING brin (created_at);
CREATE INDEX idx_trgm ON t USING gin (name gin_trgm_ops);  -- pg_trgm: LIKE '%x%' 가속
```

### 인덱스 유지보수
```sql
REINDEX INDEX CONCURRENTLY idx_x;
REINDEX TABLE CONCURRENTLY t;

-- 사용되지 않는 인덱스 찾기
SELECT relname, indexrelname, idx_scan
FROM pg_stat_user_indexes WHERE idx_scan = 0 ORDER BY relname;
```

## 5. 시퀀스

```sql
CREATE SEQUENCE seq_order_no START 1000 INCREMENT 1 CACHE 20;

SELECT nextval('seq_order_no');
SELECT currval('seq_order_no');   -- 현재 세션에서 nextval 후에만 가능
SELECT lastval();                 -- 마지막 nextval 값
SELECT setval('seq_order_no', 5000);
SELECT setval('seq_order_no', 5000, false);  -- 다음 nextval이 5000

ALTER SEQUENCE seq_order_no RESTART WITH 1;

-- IDENTITY 컬럼의 시퀀스 조작
ALTER TABLE t ALTER COLUMN id RESTART WITH 1000;
SELECT pg_get_serial_sequence('t', 'id');
```

**주의:** 시퀀스는 **트랜잭션 롤백해도 되돌아가지 않는다** (구멍 발생). 연속 번호가 필요하면
시퀀스를 쓰면 안 된다.

## 6. VIEW / MATERIALIZED VIEW

```sql
CREATE OR REPLACE VIEW v_active_users AS
SELECT id, email FROM users WHERE status = 'active';

-- CREATE OR REPLACE VIEW 는 컬럼 추가만 가능 (삭제/타입변경/순서변경 불가)
-- 구조 바꾸려면 DROP 후 재생성

-- 갱신 가능 뷰 조건: 단일 테이블, 집계/DISTINCT/GROUP BY 없음
CREATE VIEW v AS SELECT id, email FROM users WHERE status='active'
WITH CHECK OPTION;   -- 뷰 조건을 만족하는 INSERT/UPDATE만 허용

-- 구체화 뷰 (결과를 물리적으로 저장)
CREATE MATERIALIZED VIEW mv_stat AS
SELECT user_id, count(*) cnt FROM orders GROUP BY user_id
WITH DATA;

REFRESH MATERIALIZED VIEW mv_stat;
-- 무중단 갱신 (UNIQUE 인덱스 필수)
CREATE UNIQUE INDEX ON mv_stat (user_id);
REFRESH MATERIALIZED VIEW CONCURRENTLY mv_stat;
```

## 7. 파티셔닝 (선언적 파티션, PG10+)

```sql
CREATE TABLE logs (
    id bigint GENERATED ALWAYS AS IDENTITY,
    created_at timestamptz NOT NULL,
    msg text,
    PRIMARY KEY (id, created_at)   -- 파티션 키가 PK에 포함되어야 함
) PARTITION BY RANGE (created_at);

CREATE TABLE logs_2026_08 PARTITION OF logs
    FOR VALUES FROM ('2026-08-01') TO ('2026-09-01');
CREATE TABLE logs_default PARTITION OF logs DEFAULT;

-- LIST / HASH 파티션
CREATE TABLE t (region text) PARTITION BY LIST (region);
CREATE TABLE t_kr PARTITION OF t FOR VALUES IN ('KR','JP');

CREATE TABLE u (id int) PARTITION BY HASH (id);
CREATE TABLE u_0 PARTITION OF u FOR VALUES WITH (MODULUS 4, REMAINDER 0);

-- 분리/부착 (대량 삭제를 즉시 처리)
ALTER TABLE logs DETACH PARTITION logs_2026_08 CONCURRENTLY;
ALTER TABLE logs ATTACH PARTITION logs_2026_09 FOR VALUES FROM (...) TO (...);
```

## 8. DROP / TRUNCATE

```sql
DROP TABLE IF EXISTS t CASCADE;   -- CASCADE: 의존 객체(뷰, FK)까지 삭제
DROP TABLE t RESTRICT;            -- 기본값. 의존 객체 있으면 에러

TRUNCATE t;                       -- 전체 삭제 (DELETE보다 훨씬 빠름)
TRUNCATE t RESTART IDENTITY;      -- IDENTITY/serial 시퀀스도 리셋
TRUNCATE t CASCADE;               -- FK로 참조하는 테이블도 함께 TRUNCATE
TRUNCATE t1, t2;
```

**TRUNCATE는 PostgreSQL에서 트랜잭션 안전하다** (Oracle과 달리 롤백 가능).
단, `ACCESS EXCLUSIVE` 락을 잡는다.

## 9. 권한

```sql
GRANT SELECT, INSERT ON users TO app_user;
GRANT ALL ON ALL TABLES IN SCHEMA app TO app_user;
GRANT USAGE ON SCHEMA app TO app_user;
GRANT EXECUTE ON FUNCTION fn(int) TO app_user;
REVOKE DELETE ON users FROM app_user;

-- 앞으로 생성될 객체에 대한 기본 권한
ALTER DEFAULT PRIVILEGES IN SCHEMA app
    GRANT SELECT ON TABLES TO readonly_user;

-- Row Level Security (RLS) — PostgreSQL 고유
ALTER TABLE docs ENABLE ROW LEVEL SECURITY;
CREATE POLICY p_own ON docs
    USING (owner_id = current_setting('app.user_id')::bigint)
    WITH CHECK (owner_id = current_setting('app.user_id')::bigint);
```
